# 1. Create the input file automatically, minimum 20 repetitions
printf "0.8\n0.2\n20\n" > in.dat

# 2. Verify its contents
cat in.dat

# 3. Compile main.c
gcc -std=c11 -Wall -Wextra main.c -o main -lm

# 4. Run using in.dat as input and save the output to out.log
./main < in.dat > out.log

# Or in one line with the making, compiling and all of the project:
printf "0.8\n0.2\n20\n" > in.dat && gcc -std=c11 -Wall -Wextra main.c -o main.out -lm && ./main.out < in.dat > out.log

# Just compiling:
gcc -std=c11 -Wall -Wextra main.c -o main.out -lm && ./main.out
